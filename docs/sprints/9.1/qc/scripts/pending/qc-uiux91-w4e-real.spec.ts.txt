import { test, type Page } from "@playwright/test";

test.skip(!process.env.QC_REAL, "cần stack thật ep-qc: chạy bằng `QC_REAL=1 playwright test -c playwright.qc-real.config.ts`");
const PW = process.env.QC_PW ?? "Edupilot#Seed-2026";

async function login(page: Page, email: string) {
  await page.goto("/login");
  await page.locator("input[type=email]").fill(email);
  await page.locator("input[type=password]").fill(PW);
  await page.keyboard.press("Enter");
  await page.waitForURL((u) => !u.pathname.startsWith("/login"));
  await page.locator("[data-part=topbar]").waitFor();
}
const courseId = (page: Page, code: string) =>
  page.evaluate(async (cc) => {
    const a = await (await fetch("/api/v1/auth/refresh", { method: "POST", credentials: "include" })).json();
    const m = await (await fetch("/api/v1/me/courses", { headers: { authorization: `Bearer ${a.access_token}` } })).json();
    return m.items.find((i: { course: { class_code: string } }) => i.course.class_code === cc).course.id as string;
  }, code);

  test.skip(info.project.name !== "desktop");

  const bar = async () => (await page.getByRole("button", { name: /Lưu điểm danh|Đã lưu/ }).first().innerText()).replace(/\s+/g, " ");
  console.log("INFO UX12-19 thanh trước:", await bar());
  const t0 = Date.now();
  for (const name of ["Bùi Khánh Việt", "Bùi Minh Yến"]) {
    const step = Date.now();
    const rowBtn = page.getByRole("button", { name: new RegExp(name) }).first();
